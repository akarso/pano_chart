func (e *Engine) Generate(ctx Context) []insights.Signal {

    signals := []insights.Signal{}

    // Breakout pressure
    if ctx.Regime == "compression" && ctx.TransitionToTrend > 0.4 {
        signals = append(signals, insights.Signal{
            Text: "Breakout pressure building",
            Strength: ctx.TransitionToTrend,
            Type: "opportunity",
        })
    }

    // Crowding risk
    if ctx.FragilityScore > 0.7 {
        text := "High crowding risk"

        if ctx.DominantSide == "long" {
            text = "Crowded longs"
        } else if ctx.DominantSide == "short" {
            text = "Crowded shorts"
        }

        signals = append(signals, insights.Signal{
            Text: text,
            Strength: ctx.FragilityScore,
            Type: "warning",
        })
    }

    // Panic / sentiment
    if ctx.BehaviorSummary == "Panic rising" {
        signals = append(signals, insights.Signal{
            Text: "Panic increasing",
            Strength: 0.7,
            Type: "warning",
        })
    }

    // Calm / compression
    if ctx.Regime == "compression" && ctx.Volatility < 0.8 {
        signals = append(signals, insights.Signal{
            Text: "Market coiling",
            Strength: 0.6,
            Type: "info",
        })
    }

    return topN(signals, 3)
}
Ranking helper
func topN(signals []insights.Signal, n int) []insights.Signal {

    sort.Slice(signals, func(i, j int) bool {
        return signals[i].Strength > signals[j].Strength
    })

    if len(signals) > n {
        return signals[:n]
    }

    return signals
}

Aggregation logic

application/behavior/market_aggregator.go

func Aggregate(list []behavior.RetailBehavior) MarketBehavior {

    var g, f, p, pa float64

    for _, b := range list {
        g += b.Greed
        f += b.Fear
        p += b.Patience
        pa += b.Panic
    }

    count := float64(len(list))
    if count == 0 {
        return MarketBehavior{}
    }

    g /= count
    f /= count
    p /= count
    pa /= count

    return MarketBehavior{
        Greed: g,
        Fear: f,
        Patience: p,
        Panic: pa,
        Summary: summarize(g, f, p, pa),
    }
}

func (h *Handler) Insights(w http.ResponseWriter, r *http.Request) {

    symbol := mux.Vars(r)["symbol"]

    ctx := buildContext(symbol)

    signals := h.signalEngine.Generate(ctx)

    response := map[string]interface{}{
        "signals": signals,
        "behaviorSummary": ctx.BehaviorSummary,
        "fragility": ctx.FragilityScore,
    }

    json.NewEncoder(w).Encode(response)
}
