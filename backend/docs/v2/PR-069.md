void listen() {
  _iap.purchaseStream.listen((purchases) {
    for (final purchase in purchases) {
      _handlePurchase(purchase);
    }
  });
}
Handler
Future<void> _handlePurchase(PurchaseDetails purchase) async {
  if (purchase.status == PurchaseStatus.purchased) {

    await sendToBackend(purchase);

    await _iap.completePurchase(purchase);
  }
}
6️⃣ Send to Backend
Future<void> sendToBackend(PurchaseDetails purchase) async {
  await http.post(
    Uri.parse("/api/subscription/verify"),
    body: jsonEncode({
      "purchaseToken": purchase.verificationData.serverVerificationData,
      "productId": purchase.productID,
    }),
  );
}

9️⃣ Backend Endpoint
Route
POST /api/subscription/verify
Handler
func VerifyHandler(w http.ResponseWriter, r *http.Request) {

    var req struct {
        PurchaseToken string
        ProductID     string
        UserID        string
    }

    json.NewDecoder(r.Body).Decode(&req)

    sub, err := VerifySubscription(req.PurchaseToken)
    if err != nil {
        http.Error(w, "invalid", 400)
        return
    }

    saveSubscription(req.UserID, sub)

    json.NewEncoder(w).Encode(sub)
}
