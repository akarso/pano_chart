    required this.text,

      url: json['url'],
      timestamp: DateTime.fromMillisecondsSinceEpoch(json['timestamp']),
    );
  }

🧠 3️⃣ Social Feed Controller
services/social_feed_service.dart
class SocialFeedService {
  final ApiClient api;

  SocialFeedService(this.api);

  Future<List<SocialPost>?> fetch(List<String> accounts) async {
    final res = await api.post('/api/social/feed', {
      "accounts": accounts,
    });

    if (res['hasNew'] == false) {
      return null;
    }

    final posts = (res['posts'] as List)
        .map((e) => SocialPost.fromJson(e))
        .toList();

    return posts;
  }
}
Controller
class SocialFeedController extends ChangeNotifier {
  final SocialFeedService service;

  List<SocialPost> posts = [];
  bool enabled = false;

  Timer? _timer;

  SocialFeedController(this.service);

  void start(List<String> accounts) {
    if (!enabled) return;

    _timer = Timer.periodic(Duration(minutes: 1), (_) async {
      final result = await service.fetch(accounts);

      if (result != null && result.isNotEmpty) {
        posts.insertAll(0, result);
        notifyListeners();
      }
    });
  }

  void stop() {
    _timer?.cancel();
  }
}
