type Candle struct {
    OpenTime int64

    MinuteOfDay int

    Count       int
    SumMove     float64

}

type Result struct {
    Buckets []BucketResult `json:"buckets"`
}

type BucketResult struct {
    MinuteOfDay int     `json:"minute"`

    SpikeProb   float64 `json:"spike_prob"`

2️⃣ Fetch candles (Binance)
internal/volatility/fetch.go
package volatility

import (
    "encoding/json"
    "fmt"
    "net/http"
    "strconv"
)

const baseURL = "https://api.binance.com/api/v3/klines"

func FetchCandles(symbol string, startTime int64, endTime int64) ([]Candle, error) {

    url := fmt.Sprintf(
        "%s?symbol=%s&interval=1m&startTime=%d&endTime=%d&limit=1000",
        baseURL, symbol, startTime, endTime,
    )

    resp, err := http.Get(url)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    var raw [][]interface{}
    if err := json.NewDecoder(resp.Body).Decode(&raw); err != nil {
        return nil, err
    }

    candles := make([]Candle, 0, len(raw))

    for _, r := range raw {
        openTime := int64(r[0].(float64))

        open, _ := strconv.ParseFloat(r[1].(string), 64)
        high, _ := strconv.ParseFloat(r[2].(string), 64)
        low, _ := strconv.ParseFloat(r[3].(string), 64)
        closePrice, _ := strconv.ParseFloat(r[4].(string), 64)

        candles = append(candles, Candle{
            OpenTime: openTime,
            Open:     open,
            High:     high,
            Low:      low,
            Close:    closePrice,
        })
    }

    return candles, nil
}
3️⃣ Aggregation Logic
internal/volatility/aggregate.go
package volatility

import (
    "math"
    "time"
)

func Aggregate(candles []Candle) Result {

    buckets := make([]*Bucket, 1440)

    for i := 0; i < 1440; i++ {
        buckets[i] = &Bucket{
            MinuteOfDay: i,
        }
    }

    atrPeriod := 14
    atrValues := computeATR(candles, atrPeriod)

    for i := atrPeriod; i < len(candles); i++ {

        c := candles[i]
        atr := atrValues[i]

        if atr == 0 {
            continue
        }

        move := math.Abs(c.Close-c.Open) / atr

        t := time.UnixMilli(c.OpenTime).UTC()
        minute := t.Hour()*60 + t.Minute()

        b := buckets[minute]

        b.Count++
        b.SumMove += move

        if move > 1.5 {
            b.SpikeCount++
        }
    }

    // compute results
    results := make([]BucketResult, 0, 1440)

    var globalSum float64
    var globalCount int

    for _, b := range buckets {
        if b.Count == 0 {
            continue
        }
        avg := b.SumMove / float64(b.Count)
        globalSum += avg
        globalCount++
    }

    globalAvg := globalSum / float64(globalCount)

    for _, b := range buckets {

        if b.Count == 0 {
            continue
        }

        avg := b.SumMove / float64(b.Count)

        results = append(results, BucketResult{
            MinuteOfDay: b.MinuteOfDay,
            AvgMove:     avg,
            SpikeProb:   float64(b.SpikeCount) / float64(b.Count),
            Normalized:  avg / globalAvg,
        })
    }

    return Result{Buckets: results}
}
ATR helper
func computeATR(candles []Candle, period int) []float64 {

    atr := make([]float64, len(candles))

    for i := 1; i < len(candles); i++ {

        tr := math.Max(
            candles[i].High-candles[i].Low,
            math.Max(
                math.Abs(candles[i].High-candles[i-1].Close),
                math.Abs(candles[i].Low-candles[i-1].Close),
            ),
        )

        if i < period {
            atr[i] = tr
        } else {
            atr[i] = (atr[i-1]*(float64(period-1)) + tr) / float64(period)
        }
    }

    return atr
}

package volatility

import (
    "encoding/json"
    "os"
)

func SaveToFile(result Result, filename string) error {

    f, err := os.Create(filename)
    if err != nil {
        return err
    }
    defer f.Close()

    enc := json.NewEncoder(f)
    enc.SetIndent("", "  ")

    return enc.Encode(result)
}
5️⃣ Main Script
cmd/vol_aggregate/main.go
package main

import (
    "fmt"
    "time"

    "yourapp/internal/volatility"
)

func main() {

    symbol := "BTCUSDT"

    days := 150
    end := time.Now().UnixMilli()
    start := time.Now().AddDate(0, 0, -days).UnixMilli()

    var all []volatility.Candle

    current := start

    fmt.Println("Fetching candles...")

    for current < end {

        next := current + 1000*60*1000 // 1000 minutes

        candles, err := volatility.FetchCandles(symbol, current, next)
        if err != nil {
            panic(err)
        }

        if len(candles) == 0 {
            break
        }

        all = append(all, candles...)

        current = candles[len(candles)-1].OpenTime + 60000

        time.Sleep(200 * time.Millisecond) // avoid rate limits
    }

    fmt.Println("Aggregating...")

    result := volatility.Aggregate(all)

    fmt.Println("Saving...")

    if err := volatility.SaveToFile(result, "volatility_1m.json"); err != nil {
        panic(err)
    }

    fmt.Println("Done.")
}
