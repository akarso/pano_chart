## PR: feat(setups): add setup quality scoring engine

Goal: compute setup quality per token without mixing incompatible metrics.

Example API:

GET /api/token/BTCUSDT/setup?timeframe=1h

Example response:

{
  "symbol": "BTCUSDT",
  "timeframe": "1h",
  "bestSetup": "compression_breakout",
  "score": 0.78,

Core architecture

Create a Setup Engine.

application/setups/

Structure:

setups/
 ├─ engine.go
 ├─ setup_types.go

4️⃣ Compression Breakout Setup

This uses:

compression strength
volume contraction
liquidity cluster tightness

compression_setup.go

type CompressionSetup struct{}

func (s CompressionSetup) Type() setup.SetupType {
    return setup.CompressionBreakout
}

func (s CompressionSetup) Score(ctx SetupContext) float64 {

    compression := ctx.CompressionScore

    volatility := 1 - ctx.Volatility

    volume := ctx.VolumeScore

    score :=
        compression * 0.5 +
        volatility * 0.3 +
        volume * 0.2

    return clamp(score)
}

    trend := ctx.TrendScore

7️⃣ Scoring Engine

engine.go

type Engine struct {
    evaluators []SetupEvaluator
}

func NewEngine() *Engine {

    return &Engine{
        evaluators: []SetupEvaluator{
            CompressionSetup{},
            TrendSetup{},
            RangeSetup{},
        },
    }
}

func (e *Engine) Evaluate(ctx SetupContext) setup.SetupScores {

    scores := map[setup.SetupType]float64{}

    bestScore := 0.0
    bestSetup := setup.SetupType("")

    for _, ev := range e.evaluators {

        score := ev.Score(ctx)

        scores[ev.Type()] = score

        if score > bestScore {
            bestScore = score
            bestSetup = ev.Type()
        }
    }

    return setup.SetupScores{
        Symbol: ctx.Symbol,
        BestSetup: bestSetup,
        Score: bestScore,
        Scores: scores,
    }
}
8️⃣ Clamp helper

Prevent weird values.

func clamp(v float64) float64 {

    if v < 0 {
        return 0
    }

    if v > 1 {
        return 1
    }

    return v
}
