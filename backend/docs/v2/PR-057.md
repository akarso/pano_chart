3️⃣ RSS Provider (port 8081 as you described)
rss/rss_provider.go
type RSSProvider struct {
    baseURL string // e.g. http://localhost:8081
    client  *http.Client
}

func (p *RSSProvider) Platform() string {
    return "twitter"
}

func (p *RSSProvider) Fetch(account domain.Account) ([]domain.Post, error) {

    url := fmt.Sprintf("%s/%s", p.baseURL, account.Handle)

    resp, err := p.client.Get(url)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    // parse RSS (use gofeed or similar)
    feed, _ := gofeed.NewParser().Parse(resp.Body)

    posts := []domain.Post{}

    for _, item := range feed.Items {

        posts = append(posts, domain.Post{
            ID: item.GUID,
            AccountID: account.ID,
            Text: item.Title,
            URL: item.Link,
            Timestamp: item.PublishedParsed.Unix(),
        })
    }

    return posts, nil
}

5️⃣ Scheduler (rate-safe polling)
Problem you mentioned:

how to not hit rate limits but keep ~1m refresh

Solution: rotating queue
scheduler.go
type Scheduler struct {
    accounts []domain.Account
    index    int
    interval time.Duration
}
Logic
func (s *Scheduler) Next() domain.Account {

    acc := s.accounts[s.index]

    s.index = (s.index + 1) % len(s.accounts)

    return acc
}
Poll loop
func (s *Scheduler) Run(poller *Poller) {

    ticker := time.NewTicker(1 * time.Second)

    for range ticker.C {
        acc := s.Next()
        go poller.Poll(acc)
    }
}
Why this works

If:

60 accounts
1 per second

👉 full cycle = 60s ✅

7️⃣ Deduplication

func FilterNew(account domain.Account, posts []domain.Post) []domain.Post {

    result := []domain.Post{}

    for _, p := range posts {
        if p.ID == account.LastSeenPostID {
            break
        }
        result = append(result, p)
    }

    return result
}

9️⃣ Poller
poller.go
type Poller struct {
    provider providers.Provider
    cache    *Cache
    store    AccountStore
    dispatcher *Dispatcher
}

func (p *Poller) Poll(account domain.Account) {

    posts, err := p.provider.Fetch(account)
    if err != nil {
        return
    }

    newPosts := FilterNew(account, posts)

    if len(newPosts) == 0 {
        return
    }

    p.dispatcher.Dispatch(newPosts)

    account.LastSeenPostID = newPosts[0].ID
    p.store.Upsert(account)
}
