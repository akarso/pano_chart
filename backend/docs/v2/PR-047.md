## PR: feat(market): add regime history tracker
Goals

Add persistent tracking of market regimes so we can compute:

regime age
regime stability
regime transitions

New endpoint:

GET /api/market/regime/history?timeframe=4h

Example:

{
  "timeframe": "4h",
  "history": [

Why this matters

The Transition Engine becomes far more realistic.

Example:

Compression age: 4 candles → breakout probability 12%
Compression age: 32 candles → breakout probability 61%

Markets accumulate energy over time.

3️⃣ Regime Tracker

Core logic.

tracker.go

package regimehistory

import "pano/domain/market"

type Tracker struct {
    repo Repository
}

func NewTracker(repo Repository) *Tracker {
    return &Tracker{repo: repo}
}

func (t *Tracker) Update(
    timeframe string,
    newRegime market.Regime,
    timestamp int64,
) error {

    current, err := t.repo.GetLatest(timeframe)
    if err != nil {
        return err
    }

    if current == nil {

        period := market.RegimePeriod{
            Regime: newRegime,
            StartTimestamp: timestamp,
            DurationCandles: 1,
        }

        return t.repo.Append(timeframe, period)
    }

    if current.Regime == newRegime {

        current.DurationCandles++

        return t.repo.Append(timeframe, *current)
    }

    err = t.repo.CloseCurrent(timeframe, timestamp)
    if err != nil {
        return err
    }

    period := market.RegimePeriod{
        Regime: newRegime,
        StartTimestamp: timestamp,
        DurationCandles: 1,
    }

    return t.repo.Append(timeframe, period)
}

This function runs every time the regime is recalculated.

4️⃣ History Service

service.go

package regimehistory

import "pano/domain/market"

type Service struct {
    repo Repository
}

func NewService(repo Repository) *Service {
    return &Service{repo: repo}
}

func (s *Service) GetHistory(timeframe string) (market.RegimeHistory, error) {

    periods, err := s.repo.GetHistory(timeframe, 50)
    if err != nil {
        return market.RegimeHistory{}, err
    }

    age := 0

    if len(periods) > 0 {
        age = periods[len(periods)-1].DurationCandles
    }

    return market.RegimeHistory{
        Timeframe: timeframe,
        Periods: periods,
        CurrentAge: age,
    }, nil
}

7️⃣ HTTP Endpoint

market_regime_history_handler.go

func (h *Handler) History(w http.ResponseWriter, r *http.Request) {

    tf := r.URL.Query().Get("timeframe")
    if tf == "" {
        tf = "4h"
    }

    history, err := h.service.GetHistory(tf)
    if err != nil {
        http.Error(w, err.Error(), 500)
        return
    }

    json.NewEncoder(w).Encode(history)
}
