2️⃣ Market Modifier Logic
File
internal/setups/market_context.go
Core function
func MarketModifier(setup SetupScore, market MarketState) float64 {

    switch setup.Regime {

    case "uptrend", "downtrend":
        return trendMarketModifier(market)

    case "sideways":
        return sidewaysMarketModifier(market)

    case "compression":
        return compressionMarketModifier(market)

    default:
        return 1.0
    }
}
3️⃣ Trend Market Modifier
Function
func trendMarketModifier(m MarketState) float64 {

    // Strong trend environment
    if m.Effective > 0.6 {
        return 1.1
    }

    // Moderate
    if m.Effective > 0.4 {
        return 1.0
    }

    // Weakening
    if m.Effective > 0.25 {
        return 0.7
    }

    // Broken market
    return 0.4
}
Intuition
strong market → reward trend setups
weak market → punish them
4️⃣ Sideways Market Modifier
Function
func sidewaysMarketModifier(m MarketState) float64 {

    if m.Effective < 0.3 {
        return 1.1 // chaos → ranges dominate
    }

    if m.Effective < 0.5 {
        return 1.0
    }

    return 0.8 // strong trend kills sideways
}
5️⃣ Compression Market Modifier
Function
func compressionMarketModifier(m MarketState) float64 {

    // Best in transition zones

    if m.Effective > 0.4 && m.Effective < 0.6 {
        return 1.1
    }

    if m.Effective < 0.3 {
        return 0.9
    }

    if m.Effective > 0.7 {
        return 0.9
    }

    return 1.0
}
