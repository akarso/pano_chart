2️⃣ Engine (central dispatcher)
engine.go
package notifications

import (
    "time"
)

type Engine struct {
    sender Sender
    dedup  Deduplicator
}

type Sender interface {
    Send(n Notification) error
}

func (e *Engine) Send(n Notification) error {

    if !withinAllowedHours() {
        return nil
    }

    if e.dedup.Seen(n.Key) {
        return nil
    }

    e.dedup.Mark(n.Key)

    return e.sender.Send(n)
}

func withinAllowedHours() bool {
    now := time.Now()
    hour := now.Hour()

    return hour >= 7 && hour <= 22
}
3️⃣ Dedup (critical)
dedup.go
package notifications

import "sync"

type Deduplicator struct {
    seen map[string]bool
    mu   sync.Mutex
}

func NewDedup() *Deduplicator {
    return &Deduplicator{
        seen: make(map[string]bool),
    }
}

func (d *Deduplicator) Seen(key string) bool {
    d.mu.Lock()
    defer d.mu.Unlock()
    return d.seen[key]
}

func (d *Deduplicator) Mark(key string) {
    d.mu.Lock()
    defer d.mu.Unlock()
    d.seen[key] = true
}

4️⃣ News CLI script
cmd/notify_news/main.go
package main

import (
    "flag"
    "fmt"
    "os"
    "strings"

    "yourapp/internal/notifications"
)

func main() {

    file := flag.String("file", "", "markdown file")
    msg := flag.String("msg", "", "direct message")
    flag.Parse()

    var content string

    if *file != "" {
        data, err := os.ReadFile(*file)
        if err != nil {
            panic(err)
        }

        content = extractExcerpt(string(data))
    } else {
        content = *msg
    }

    engine := buildEngine()

    n := notifications.Notification{
        Type:  notifications.TypeNews,
        Title: "Market News",
        Body:  content,
        Key:   "news_" + content,
    }

    if err := engine.Send(n); err != nil {
        panic(err)
    }

    fmt.Println("Sent")
}

func extractExcerpt(md string) string {
    lines := strings.Split(md, "\n")

    if len(lines) > 3 {
        return strings.Join(lines[:3], " ")
    }
    return md
}

func (e *Engine) CheckMacroEvents(events []Event) {

    now := time.Now()

    for _, ev := range events {

        diff := ev.Time.Sub(now)

        if diff > 29*time.Minute && diff < 31*time.Minute {

            key := "macro_" + ev.ID

            e.Send(Notification{
                Type:  TypeMacro,
                Title: "Upcoming Macro Event",
                Body:  ev.Name + " in 30 minutes",
                Key:   key,
            })
        }
    }
}

8️⃣ Once-per-day helper
var lastRun = make(map[string]time.Time)

func isOncePerDay(key string) bool {
    now := time.Now()

    last, ok := lastRun[key]
    if !ok || now.Sub(last) > 24*time.Hour {
        lastRun[key] = now
        return true
    }

    return false
}
