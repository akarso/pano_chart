1️⃣ Backend — Trend Health Layer
New file

Core model
type TrendHealth struct {
    Value float64 // 0.0 - 1.0
}
Function
func ComputeTrendHealth(
    state string, // "uptrend" / "downtrend"
    price float64,
    recentHigh float64,
    recentLow float64,
    atr float64,
    recentReturn float64, // in ATR units
) float64 {

    if atr == 0 {
        return 0
    }

    var health float64

    switch state {

    case "uptrend":
        drawdown := (recentHigh - price) / atr
        health = 1.0 - clamp(drawdown, 0, 1)

    case "downtrend":
        bounce := (price - recentLow) / atr
        health = 1.0 - clamp(bounce, 0, 1)

    default:
        return 0
    }

    // Crash penalty
    if recentReturn < -1.5 {
        health *= 0.3
    }

    return clamp(health, 0, 1)
}
Helper
func clamp(v, min, max float64) float64 {
    if v < min {
        return min
    }
    if v > max {
        return max
    }
    return v
}
2️⃣ Backend — Effective Aggregation
Update market aggregation
type MarketStats struct {
    TrendPrevalence float64
    EffectiveTrend  float64
    BreakdownRate   float64
}
Aggregation logic
func AggregateMarket(tokens []TokenState) MarketStats {

    var trendCount float64
    var effectiveSum float64
    var breakdowns float64

    for _, t := range tokens {

        if t.State != "uptrend" && t.State != "downtrend" {
            continue
        }

        trendCount++

        health := t.TrendHealth
        effectiveSum += health

        if health < 0.4 {
            breakdowns++
        }
    }

    total := float64(len(tokens))

    return MarketStats{
        TrendPrevalence: trendCount / total,
        EffectiveTrend:  effectiveSum / total,
        BreakdownRate:   breakdowns / total,
    }
}

Label logic
func BuildMarketLabel(s MarketStats) string {

    if s.TrendPrevalence > 0.6 {

        if s.Effective > 0.5 {
            return "Strong trend"
        }

        if s.Effective > 0.3 {
            return "Trend weakening"
        }

        return "Trend breaking down"
    }

    if s.TrendPrevalence > 0.4 {
        return "Mixed conditions"
    }

    return "No clear trend"
}
4️⃣ Backend — Push Notifications
Update trigger logic
Market notification (once per day)
func (e *Engine) CheckMarketState(state MarketState) {

    if !isOncePerDay("market_state") {
        return
    }

    if state.Prevalence < 0.6 {
        return
    }

    msg := state.Label

    e.Send(Notification{
        Type:  TypeMarket,
        Title: "Market Update",
        Body:  msg,
        Key:   "market_" + state.Label,
    })
}
