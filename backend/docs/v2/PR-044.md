2️⃣ Application Layer

Create metrics service.

application/market/metrics/composite_index_service.go

package metrics

import (
    "sort"

    "pano/domain/market"
    "pano/domain/marketdata"
)

type CandleProvider interface {
    GetCandles(symbol string, timeframe string, limit int) ([]marketdata.Candle, error)
    GetSymbols() []string
}

type CompositeIndexService struct {
    provider CandleProvider
}

func NewCompositeIndexService(p CandleProvider) *CompositeIndexService {
    return &CompositeIndexService{provider: p}
}

func (s *CompositeIndexService) Calculate(timeframe string, limit int) (market.CompositeIndex, error) {

    symbols := s.provider.GetSymbols()

    normalizedSeries := [][]float64{}
    timestamps := []int64{}

    for _, sym := range symbols {

        candles, err := s.provider.GetCandles(sym, timeframe, limit)
        if err != nil || len(candles) == 0 {
            continue
        }

        base := candles[0].Close

        series := make([]float64, len(candles))

        for i, c := range candles {
            series[i] = (c.Close / base) * 100
        }

        normalizedSeries = append(normalizedSeries, series)

        if len(timestamps) == 0 {
            for _, c := range candles {
                timestamps = append(timestamps, c.Timestamp)
            }
        }
    }

    points := []market.IndexPoint{}

    for i := range timestamps {

        values := []float64{}

        for _, series := range normalizedSeries {
            if i < len(series) {
                values = append(values, series[i])
            }
        }

        if len(values) == 0 {
            continue
        }

        sort.Float64s(values)

        median := values[len(values)/2]

        points = append(points, market.IndexPoint{
            Timestamp: timestamps[i],
            Value:     median,
        })
    }

    return market.CompositeIndex{
        Timeframe:   timeframe,
        Points:      points,
        SymbolCount: len(symbols),
    }, nil
}
3️⃣ Why Median Instead of Mean

Crypto outliers distort averages.

Example:

149 tokens flat
1 token +40%

Mean becomes misleading.

Median gives true central movement.

This is standard in robust statistics.

4️⃣ HTTP Adapter

adapters/http/market_composite_handler.go

package http

import (
    "encoding/json"
    "net/http"
    "strconv"

    metrics "pano/application/market/metrics"
)

type MarketCompositeHandler struct {
    service *metrics.CompositeIndexService
}

func NewMarketCompositeHandler(s *metrics.CompositeIndexService) *MarketCompositeHandler {
    return &MarketCompositeHandler{service: s}
}

func (h *MarketCompositeHandler) Composite(w http.ResponseWriter, r *http.Request) {

    tf := r.URL.Query().Get("timeframe")
    if tf == "" {
        tf = "4h"
    }

    limitStr := r.URL.Query().Get("limit")
    limit := 200

    if limitStr != "" {
        if v, err := strconv.Atoi(limitStr); err == nil {
            limit = v
        }
    }

    index, err := h.service.Calculate(tf, limit)
    if err != nil {
        http.Error(w, err.Error(), http.StatusInternalServerError)
        return
    }

    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(index)
}
