## PR: feat(market): add transition probability engine

Goal: expose

GET /api/market/transition?timeframe=4h

Example response:

{
  "timeframe": "4h",
  "currentRegime": "compression",
  "probabilities": {
    "trend": 0.28,
    "sideways": 0.52,
    "expansion": 0.20
  },
  "horizon": "1h"
}

Important:
This is not ML. It is a probabilistic heuristic engine based on market conditions.

Core idea

Transitions depend on a few measurable variables:

compression breadth
volatility slope
market dispersion
regime age

These variables determine pressure build-up.

2️⃣ Pressure Model

This is the core concept.

We calculate market pressure:

breakoutPressure =
    compressionBreadth
  * volatilitySlope
  * regimeAgeFactor

Implementation:

pressure_model.go

func breakoutPressure(
    compressionBreadth float64,
    volatilitySlope float64,
    regimeAge float64,
) float64 {

    pressure := compressionBreadth *
        (1 + volatilitySlope) *
        regimeAge

    if pressure > 1 {
        pressure = 1
    }

    return pressure
}
3️⃣ Regime Age

Markets staying longer in compression increase breakout probability.

Example:

compression age 5 candles → weak
compression age 30 candles → strong

Implementation:

func regimeAgeFactor(age int) float64 {

    if age >= 30 {
        return 1
    }

    return float64(age) / 30
}
4️⃣ Volatility Slope

Detect if volatility is starting to increase.

func volatilitySlope(series []float64) float64 {

    if len(series) < 5 {
        return 0
    }

    start := series[len(series)-5]
    end := series[len(series)-1]

    return (end - start) / start
}

Interpretation:

>0 volatility rising
<0 volatility compressing
5️⃣ Transition Engine

transition_engine.go

type TransitionEngine struct{}

func (e *TransitionEngine) Calculate(
    regime market.Regime,
    compressionBreadth float64,
    volatilitySlope float64,
    regimeAge int,
) market.TransitionProbabilities {

    ageFactor := regimeAgeFactor(regimeAge)

    pressure := breakoutPressure(
        compressionBreadth,
        volatilitySlope,
        ageFactor,
    )

    switch regime {

    case market.RegimeCompression:

        trend := pressure * 0.6
        expansion := pressure * 0.4
        sideways := 1 - (trend + expansion)

        return market.TransitionProbabilities{
            Trend: trend,
            Expansion: expansion,
            Sideways: sideways,
        }

    case market.RegimeTrend:

        continuation := 0.6 + volatilitySlope
        if continuation > 0.9 {
            continuation = 0.9
        }

        return market.TransitionProbabilities{
            Trend: continuation,
            Sideways: 1 - continuation,
            Expansion: 0,
        }

    default:

        return market.TransitionProbabilities{
            Trend: 0.3,
            Sideways: 0.5,
            Expansion: 0.2,
        }
    }
}
6️⃣ Transition Service

transition_service.go

type TransitionService struct {
    regimeService *metrics.MetricsService
    engine        *TransitionEngine
}

func (s *TransitionService) Calculate(timeframe string) (market.MarketTransition, error) {

    regimeSummary, err := s.regimeService.CalculateRegime(timeframe)
    if err != nil {
        return market.MarketTransition{}, err
    }

    regimeAge := 12 // TODO: derive from regime history

    volatilitySlope := regimeSummary.Metrics.VolatilityExpansion - 1

    probs := s.engine.Calculate(
        regimeSummary.Regime,
        regimeSummary.Metrics.CompressionBreadth,
        volatilitySlope,
        regimeAge,
    )

    return market.MarketTransition{
        Timeframe: timeframe,
        CurrentRegime: regimeSummary.Regime,
        Probabilities: probs,
        Horizon: "1h",
    }, nil
}

7️⃣ HTTP Adapter

market_transition_handler.go

func (h *MarketTransitionHandler) Transition(w http.ResponseWriter, r *http.Request) {

    tf := r.URL.Query().Get("timeframe")
    if tf == "" {
        tf = "4h"
    }

    result, err := h.service.Calculate(tf)
    if err != nil {
        http.Error(w, err.Error(), 500)
        return
    }

    json.NewEncoder(w).Encode(result)
}
