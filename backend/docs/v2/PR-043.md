## PR-043: feat(market): add market state + breadth aggregation
Goal

Expose global market structure summary derived from existing per-symbol evaluations.

This adds:

GET /api/market/state?timeframe=4h

Response:

{
  "timeframe": "4h",
  "state": "sideways",
  "confidence": 0.54,
  "breadth": {
    "sideways": 0.54,
    "compression": 0.22,
    "trend": 0.16,
    "breakout": 0.08
  },
  "symbolCount": 150
}

No changes to scoring algorithms.

1️⃣ Domain Layer

Create a pure value object.

domain/market/market_state.go

package market

type State string

const (
    StateSideways    State = "sideways"
    StateCompression State = "compression"
    StateBreakout    State = "breakout"
    StateTrend       State = "trend"
)

type Breadth struct {
    Sideways    float64
    Compression float64
    Breakout    float64
    Trend       float64
}

type Summary struct {
    Timeframe   string
    State       State
    Confidence  float64
    Breadth     Breadth
    SymbolCount int
}

No dependencies outside domain.

2️⃣ Application Layer

Create use case service.

application/market/market_state_service.go

package market

import (
    "pano/domain/market"
    "pano/domain/scoring"
)

type EvaluationProvider interface {
    GetLatestEvaluations(timeframe string) ([]scoring.EvaluationSnapshot, error)
}

type MarketStateService struct {
    provider EvaluationProvider
}

func NewMarketStateService(p EvaluationProvider) *MarketStateService {
    return &MarketStateService{provider: p}
}

func (s *MarketStateService) Calculate(timeframe string) (market.Summary, error) {

    evaluations, err := s.provider.GetLatestEvaluations(timeframe)
    if err != nil {
        return market.Summary{}, err
    }

    counts := map[market.State]int{
        market.StateSideways:    0,
        market.StateCompression: 0,
        market.StateBreakout:    0,
        market.StateTrend:       0,
    }

    for _, e := range evaluations {

        state := classify(e)

        counts[state]++
    }

    total := float64(len(evaluations))

    breadth := market.Breadth{
        Sideways:    float64(counts[market.StateSideways]) / total,
        Compression: float64(counts[market.StateCompression]) / total,
        Breakout:    float64(counts[market.StateBreakout]) / total,
        Trend:       float64(counts[market.StateTrend]) / total,
    }

    dominant := market.StateSideways
    max := 0

    for s, c := range counts {
        if c > max {
            max = c
            dominant = s
        }
    }

    confidence := float64(max) / total

    return market.Summary{
        Timeframe:   timeframe,
        State:       dominant,
        Confidence:  confidence,
        Breadth:     breadth,
        SymbolCount: len(evaluations),
    }, nil
}
3️⃣ Classification Logic

Still in application layer.

application/market/state_classifier.go

package market

import (
    "pano/domain/market"
    "pano/domain/scoring"
)

func classify(e scoring.EvaluationSnapshot) market.State {

    if e.BreakoutUp > 0.7 || e.BreakoutDown > 0.7 {
        return market.StateBreakout
    }

    if e.CompressionScore > 0.7 {
        return market.StateCompression
    }

    if e.TrendScore > 0.65 {
        return market.StateTrend
    }

    return market.StateSideways
}

Thresholds intentionally conservative.

4️⃣ Adapter: HTTP Handler

adapters/http/market_handler.go

package http

import (
    "encoding/json"
    "net/http"

    app "pano/application/market"
)

type MarketHandler struct {
    service *app.MarketStateService
}

func NewMarketHandler(s *app.MarketStateService) *MarketHandler {
    return &MarketHandler{service: s}
}

func (h *MarketHandler) MarketState(w http.ResponseWriter, r *http.Request) {

    tf := r.URL.Query().Get("timeframe")
    if tf == "" {
        tf = "4h"
    }

    summary, err := h.service.Calculate(tf)
    if err != nil {
        http.Error(w, err.Error(), http.StatusInternalServerError)
        return
    }

    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(summary)
}
