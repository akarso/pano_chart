2️⃣ Access Policy (single source of truth)
application/billing/feature_policy.go
type FeaturePolicy struct{}

func (p *FeaturePolicy) Has(plan billing.Plan, feature billing.Feature) bool {

    isPro := plan.IsPro()

    switch feature {

    case billing.FeatureTokensLimit:
        return true // handled separately (value-based)

    case billing.FeatureAutoRefresh:
        return isPro

    case billing.FeatureAdvancedSorting:
        return isPro

    case billing.FeatureFutureXAxis:
        return isPro

    case billing.FeatureMarketState:
        return isPro

    case billing.FeatureMarketComposite:
        return isPro

    case billing.FeatureMarketBreadth:

6️⃣ Flutter: Feature Gate Service
feature_gate.dart
class FeatureGate {
  final Plan plan;

  FeatureGate(this.plan);

  bool get isPro => plan == Plan.pro || plan == Plan.trial;

  bool has(Feature f) {
    switch (f) {
      case Feature.autoRefresh:
      case Feature.marketState:
      case Feature.signals:
      case Feature.retailBehavior:
        return isPro;
      default:
        return false;
    }
  }

  int get tokenLimit => isPro ? 150 : 10;
}

8️⃣ Trial Handling
Backend
func getPlan(r *http.Request) billing.Plan {

    user := getUser(r)

    if user.TrialEndsAt.After(time.Now()) {
        return billing.PlanTrial
    }

    if user.HasActiveSubscription {
        return billing.PlanPro
    }

    return billing.PlanFree
}
