class NotificationSettings {

  bool macro;
  bool marketRegime;
  bool setupOfDay;
  bool news;

  NotificationSettings({
    required this.social,
    required this.macro,
    required this.marketRegime,
    required this.setupOfDay,
    required this.news,
  });

      marketRegime: true,

2️⃣ Settings Page UI
notification_settings_page.dart

Keep it flat, fast, no nesting.

import 'package:flutter/material.dart';

class NotificationSettingsPage extends StatefulWidget {
  const NotificationSettingsPage({super.key});

  @override
  State<NotificationSettingsPage> createState() =>
      _NotificationSettingsPageState();
}

class _NotificationSettingsPageState extends State<NotificationSettingsPage> {
  late NotificationSettings settings;

  @override
  void initState() {
    super.initState();
    settings = NotificationSettings.defaults(); // replace with persisted
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text("Notifications")),
      body: ListView(
        children: [
          _sectionTitle("Social"),
          _toggle(
            "Twitter Alerts",
            settings.social,
            (v) => setState(() => settings.social = v),
          ),

          _sectionTitle("Market"),
          _toggle(
            "Market Regime",
            settings.marketRegime,
            (v) => setState(() => settings.marketRegime = v),
          ),
          _toggle(
            "Setup of the Day",
            settings.setupOfDay,
            (v) => setState(() => settings.setupOfDay = v),
          ),

          _sectionTitle("Macro"),
          _toggle(
            "Macro Events (30m before)",
            settings.macro,
            (v) => setState(() => settings.macro = v),
          ),

          _sectionTitle("News"),
          _toggle(
            "News Alerts",
            settings.news,
            (v) => setState(() => settings.news = v),
          ),
        ],
      ),
    );
  }

  Widget _sectionTitle(String title) {
    return Padding(
      padding: const EdgeInsets.fromLTRB(16, 20, 16, 8),
      child: Text(
        title,
        style: const TextStyle(fontWeight: FontWeight.bold),
      ),
    );
  }

  Widget _toggle(String label, bool value, Function(bool) onChanged) {
    return SwitchListTile(
      title: Text(label),
      value: value,
      onChanged: onChanged,
    );
  }
}

3️⃣ Deep link routing (core part)
notification_router.dart

This is where the magic happens.

import 'package:flutter/material.dart';

class NotificationRouter {
  static void handle(Map<String, dynamic> data, BuildContext context) {
    final type = data['type'];

    switch (type) {
      case 'twitter':
        Navigator.pushNamed(context, '/social');
        break;

      case 'macro':
        Navigator.pushNamed(context, '/events');
        break;

      case 'news':
        Navigator.pushNamed(context, '/news');
        break;

      case 'market':
        Navigator.pushNamed(context, '/market');
        break;

      case 'setup':
        final symbol = data['symbol'];
        Navigator.pushNamed(context, '/token', arguments: symbol);
        break;

      default:
        Navigator.pushNamed(context, '/');
    }
  }
}
