import 'package:flutter/material.dart';
import 'volatility_model.dart';
import 'volatility_painter.dart';

class VolatilityWidget extends StatelessWidget {
  final List<VolatilityBucket> data;
  final int visibleStartMinute; // based on chart viewport
  final int visibleCount;

  const VolatilityWidget({
    super.key,
    required this.data,
    required this.visibleStartMinute,
    required this.visibleCount,
  });

  @override
  Widget build(BuildContext context) {
    return SizedBox(
      height: 60, // compact, like volume
      child: CustomPaint(
        painter: VolatilityPainter(
          data: data,
          start: visibleStartMinute,
          count: visibleCount,
        ),
      ),
    );
  }
}
3️⃣ Painter (core logic)
volatility_painter.dart
import 'package:flutter/material.dart';
import 'volatility_model.dart';
import 'dart:math';

class VolatilityPainter extends CustomPainter {
  final List<VolatilityBucket> data;
  final int start;
  final int count;

  VolatilityPainter({
    required this.data,
    required this.start,
    required this.count,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final barWidth = size.width / count;
    final baselineY = size.height / 2;

    final paint = Paint();

    for (int i = 0; i < count; i++) {
      final idx = (start + i) % data.length;
      final b = data[idx];

      final x = i * barWidth;

      // Normalize around 1.0 baseline
      final deviation = b.normalized - 1.0;

      // scale factor (tune this!)
      final height = deviation * size.height * 0.8;

      final top = deviation >= 0
          ? baselineY - height
          : baselineY;

      final bottom = deviation >= 0
          ? baselineY
          : baselineY - height;

      paint.color = _colorFromSpike(b.spikeProb);

      canvas.drawRect(
        Rect.fromLTRB(x, top, x + barWidth * 0.9, bottom),
        paint,
      );
    }

    // Draw baseline
    final baselinePaint = Paint()
      ..color = Colors.white.withOpacity(0.2)
      ..strokeWidth = 1;

    canvas.drawLine(
      Offset(0, baselineY),
      Offset(size.width, baselineY),
      baselinePaint,
    );
  }

  Color _colorFromSpike(double spike) {
    // clamp
    spike = spike.clamp(0.0, 1.0);

    if (spike < 0.33) {
      return Colors.green.withOpacity(0.8);
    } else if (spike < 0.66) {
      return Colors.yellow.withOpacity(0.85);
    } else {
      return Colors.red.withOpacity(0.9);
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => true;
}
🎨 Visual behavior
Baseline logic
normalized = 1.0 → middle line
>1.0 → above (more active)
<1.0 → below (quiet)
Color meaning
Color	Meaning
🟢 green	calm moves
🟡 yellow	moderate spikes
🔴 red	high spike probability
